💬 3. Muro de Peticiones

Escribe que banda o artista no puede faltar en el setlist final.

// EJ3 // Obtener los elementos del DOM const inputPeticion = document.querySelector("#input-peticion"); const btnEnviar = document.querySelector("#btn-enviar"); const muro = document.querySelector("#muro"); // Escuchador de eventos btnEnviar.addEventListener("click", function() { // Obtener el valor del input const nombreArtista = inputPeticion.value.trim(); // Verificar que no esté vacío if (nombreArtista !== "") { // Crear un nuevo elemento const nuevaPeticion = document.createElement("li"); // Añadir clases de Bootstrap nuevaPeticion.classList.add("list-group-item", "bg-dark", "text-light", "border-secondary"); // Agregar el nombre del artista nuevaPeticion.textContent = "🎵 " + nombreArtista; // Insertar la petición en el muro muro.appendChild(nuevaPeticion); // Limpiar el input inputPeticion.value = ""; } else { alert("Por favor, escribe el nombre de un artista o banda."); } });
